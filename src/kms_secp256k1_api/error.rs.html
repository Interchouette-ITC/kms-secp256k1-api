1//! Domain errors for KMS API services and HTTP handlers.
2//!
3//! Library / service boundary uses [`KmsError`] (`thiserror`). HTTP handlers map
4//! variants to status via [`KmsError::status`] and `Display` into JSON `{"error": ...}`.
5
6use hyper::StatusCode;
7use thiserror::Error;
8
9/// Crate result alias for service and KMS client code.
10pub type Result<T> = std::result::Result<T, KmsError>;
11
12/// Typed failures from key ops, signing, verification, AWS KMS, and crypto/WASM.
13///
14/// `# Errors` rustdoc on `pub` APIs should name the relevant variants / conditions.
15#[derive(Debug, Error)]
16pub enum KmsError {
17    /// Wrong blockchain / feature mode for the requested operation.
18    #[error("Only {mode} mode is supported")]
19    ModeMismatch {
20        /// Expected mode name (e.g. `Ethereum`, `Casper`, `Cosmos`).
21        mode: &'static str,
22    },
23
24    /// Production path requires a supported mode (or testing).
25    #[error("Unsupported KMS mode and not in testing")]
26    UnsupportedMode,
27
28    /// Alias / address / public key lookup failed in local store or KMS.
29    #[error("Key not found")]
30    KeyNotFound,
31
32    /// Public key missing from keystore entry.
33    #[error("Public key not found")]
34    PublicKeyNotFound,
35
36    /// AWS alias metadata missing.
37    #[error("KeyMetadata not found for alias {alias}")]
38    AliasNotFound {
39        /// Alias that was looked up.
40        alias: String,
41    },
42
43    /// KMS response lacked a key id.
44    #[error("No KeyId found")]
45    MissingKeyId,
46
47    /// Key creation produced an empty public key.
48    #[error("No public key generated")]
49    EmptyPublicKey,
50
51    /// Hex decode / format failure.
52    #[error("Invalid hex: {0}")]
53    InvalidHex(String),
54
55    /// Transaction hash hex is invalid.
56    #[error("Invalid transaction hash hex: {0}")]
57    InvalidTxHashHex(String),
58
59    /// Transaction hash is not 32 bytes.
60    #[error("Transaction hash must be 32 bytes")]
61    TxHashWrongLength,
62
63    /// Signature bytes failed hex decode.
64    #[error("Invalid hex in signature")]
65    InvalidSignatureHex,
66
67    /// Signature length is not the expected size.
68    #[error("Invalid signature length (expected {expected} bytes)")]
69    InvalidSignatureLength {
70        /// Expected byte length.
71        expected: usize,
72    },
73
74    /// Request or intermediate JSON failed to parse.
75    #[error("Failed to parse input JSON: {0}")]
76    ParseJson(String),
77
78    /// Transaction body / params parse failure (chain-specific wording preserved in message).
79    #[error("{0}")]
80    ParseTransaction(String),
81
82    /// Transaction shape is not supported.
83    #[error("Unsupported transaction format")]
84    UnsupportedTxFormat,
85
86    /// Signing failed after inputs were accepted.
87    #[error("Signing failed: {0}")]
88    SigningFailed(String),
89
90    /// Local or WASM verification returned false / failed.
91    #[error("Signature verification failed")]
92    VerificationFailed,
93
94    /// Verification failed with detail.
95    #[error("Signature verification failed: {0}")]
96    VerificationFailedDetail(String),
97
98    /// Fresh signature failed self-check.
99    #[error("Generated signature failed verification")]
100    PostSignVerifyFailed,
101
102    /// AWS KMS (or mock client) operation failed.
103    #[error("{0}")]
104    Kms(String),
105
106    /// Crypto / WASM helper failure.
107    #[error("{0}")]
108    Crypto(String),
109
110    /// Cosmos REST / account / proto pipeline failure.
111    #[error("{0}")]
112    Cosmos(String),
113
114    /// Catch-all for remaining stringly errors; prefer a named variant when touching call sites.
115    #[error("{0}")]
116    Msg(String),
117}
118
119impl KmsError {
120    /// Build [`KmsError::Msg`] from any displayable value.
121    #[must_use]
122    pub fn msg(s: impl Into<String>) -> Self {
123        Self::Msg(s.into())
124    }
125
126    /// Map a displayable failure into [`KmsError::Crypto`].
127    #[must_use]
128    pub fn from_dyn(err: impl std::fmt::Display) -> Self {
129        Self::Crypto(err.to_string())
130    }
131
132    /// HTTP status for this failure (client vs server).
133    #[must_use]
134    pub const fn status(&self) -> StatusCode {
135        match self {
136            Self::KeyNotFound | Self::PublicKeyNotFound | Self::AliasNotFound { .. } => {
137                StatusCode::NOT_FOUND
138            }
139            Self::ModeMismatch { .. }
140            | Self::UnsupportedMode
141            | Self::InvalidHex(_)
142            | Self::InvalidTxHashHex(_)
143            | Self::TxHashWrongLength
144            | Self::InvalidSignatureHex
145            | Self::InvalidSignatureLength { .. }
146            | Self::ParseJson(_)
147            | Self::ParseTransaction(_)
148            | Self::UnsupportedTxFormat
149            | Self::VerificationFailed
150            | Self::VerificationFailedDetail(_) => StatusCode::BAD_REQUEST,
151            Self::MissingKeyId
152            | Self::EmptyPublicKey
153            | Self::SigningFailed(_)
154            | Self::PostSignVerifyFailed
155            | Self::Kms(_)
156            | Self::Crypto(_)
157            | Self::Cosmos(_)
158            | Self::Msg(_) => StatusCode::INTERNAL_SERVER_ERROR,
159        }
160    }
161}

163impl From<String> for KmsError {
164    fn from(value: String) -> Self {
165        Self::Msg(value)
166    }
